```ts
// src/hooks/useDashboardKpis.ts
import { useMemo } from 'react';
import { useBalanceStore } from '../stores/useBalanceStore';
import { useTemplateStore } from '../stores/useTemplateStore';
import { useCategoryStore } from '../stores/useCategoryStore';
import { useMonthlyStore } from '../stores/useMonthlyStore';
import { generateForecast } from '../utils/forecast';
import { toYearMonth } from '../types/ui';

export interface DashboardKpis {
  thisMonthNet: number;
  minBalance90d: number;
  nextLargeExpense: { date: string; name: string; amount: number } | null;
  slopePerDay: number;
}

export const useDashboardKpis = (): DashboardKpis => {
  const balance = useBalanceStore((s) => s.balance);
  const templates = useTemplateStore((s) => s.templates);
  const categories = useCategoryStore((s) => s.categories);
  const monthlyAmountsMap = useMonthlyStore((s) => s.monthlyAmountsMap);

  return useMemo<DashboardKpis>(() => {
    const today = new Date();
    const horizon = new Date(today);
    horizon.setDate(horizon.getDate() + 90);

    const points = generateForecast({
      startBalance: balance,
      startDate: today,
      endDate: horizon,
      templates,
      categories,
      monthlyAmountsMap,
    });

    const thisMonth = toYearMonth(today);
    let income = 0;
    let expense = 0;
    for (const p of points) {
      if (!p.date.startsWith(thisMonth)) continue;
      for (const ev of p.eventDetails) {
        if (ev.type === 'income') income += ev.amount;
        else expense += ev.amount;
      }
    }

    let min = Infinity;
    for (const p of points) {
      if (p.balance < min) min = p.balance;
    }

    let nextLarge: DashboardKpis['nextLargeExpense'] = null;
    let maxAmount = 0;
    const next60 = new Date(today);
    next60.setDate(next60.getDate() + 60);
    for (const p of points) {
      const d = new Date(p.date);
      if (d > next60) break;
      for (const ev of p.eventDetails) {
        if (ev.type === 'expense' && ev.amount > maxAmount) {
          maxAmount = ev.amount;
          nextLarge = { date: p.date, name: ev.name, amount: ev.amount };
        }
      }
    }

    const first = points[0]?.balance ?? balance;
    const last = points[points.length - 1]?.balance ?? balance;
    const slope = points.length > 1 ? (last - first) / points.length : 0;

    return {
      thisMonthNet: income - expense,
      minBalance90d: min === Infinity ? balance : min,
      nextLargeExpense: nextLarge,
      slopePerDay: slope,
    };
  }, [balance, templates, categories, monthlyAmountsMap]);
};
```

Adjust `generateForecast` args to match the actual signature in `src/utils/forecast.ts` — read that file first. If names differ, adapt.

// src/components/dashboard/KpiHero.tsx
import CountUp from 'react-countup';
import { Card } from '../ui/Card';
import { Badge } from '../ui/Badge';
import { useDashboardKpis } from '../../hooks/useDashboardKpis';

const formatYen = (v: number): string => `¥${Math.round(v).toLocaleString()}`;

export function KpiHero(): React.ReactElement {
  const kpis = useDashboardKpis();
  const netTone: 'success' | 'danger' | 'neutral' =

        </div>
        <Badge tone={netTone} className="mt-2">

      </Card>

      <Card padding="md" glow="blue">
        <div className="text-xs font-medium text-content-muted">最小残高（90日）</div>
        <div className="mt-2 text-2xl font-semibold text-content-primary tabular-nums">
          <CountUp end={kpis.minBalance90d} duration={0.6} formattingFn={formatYen} />
        </div>
        <Badge tone={kpis.minBalance90d < 0 ? 'danger' : 'info'} className="mt-2">

        </Badge>
      </Card>

      <Card padding="md">
        <div className="text-xs font-medium text-content-muted">次の大型支出（60日）</div>

          </>

          {kpis.slopePerDay >= 0 ? '増加傾向' : '減少傾向'}

  );
}

**Files:**

- Modify: `src/App.tsx`

- [ ] **Step 1: Create useKeyboardShortcuts**

```ts
// src/hooks/useKeyboardShortcuts.ts
import { useEffect } from 'react';
import type { ViewType } from '../types';
import { useUIStore } from '../stores/useUIStore';

interface UseKeyboardShortcutsArgs {
  onNavigate: (v: ViewType) => void;
  onShowHelp: () => void;
}

const isEditable = (el: EventTarget | null): boolean => {
  if (!(el instanceof HTMLElement)) return false;
  const tag = el.tagName;
  return tag === 'INPUT' || tag === 'TEXTAREA' || tag === 'SELECT' || el.isContentEditable;
};

export const useKeyboardShortcuts = ({ onNavigate, onShowHelp }: UseKeyboardShortcutsArgs): void => {
  const toggleSidebar = useUIStore((s) => s.toggleSidebar);

  useEffect(() => {
    let awaitingG = false;
    let gTimer: ReturnType<typeof setTimeout> | null = null;

    const resetG = (): void => {
      awaitingG = false;
      if (gTimer) { clearTimeout(gTimer); gTimer = null; }
    };

    const handle = (e: KeyboardEvent): void => {
      if (isEditable(e.target)) return;

      if ((e.metaKey || e.ctrlKey) && e.key === '\\') {
        e.preventDefault();
        toggleSidebar();
        return;
      }

      if (e.key === '?') {
        e.preventDefault();
        onShowHelp();
        return;
      }

      if (awaitingG) {
        const map: Record<string, ViewType> = {
          d: 'dashboard',
          e: 'entries',
          h: 'history',
          a: 'analytics',
          s: 'settings',
        };
        const view = map[e.key.toLowerCase()];
        if (view) {
          e.preventDefault();
          onNavigate(view);
        }
        resetG();
        return;
      }

      if (e.key.toLowerCase() === 'g' && !e.metaKey && !e.ctrlKey && !e.altKey) {
        awaitingG = true;
        gTimer = setTimeout(resetG, 1200);
      }
    };

    document.addEventListener('keydown', handle);
    return () => {
      document.removeEventListener('keydown', handle);
      resetG();
    };
  }, [onNavigate, onShowHelp, toggleSidebar]);
};
```
